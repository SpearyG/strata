cmake_minimum_required(VERSION 3.30)
project(strata_audit_measurement LANGUAGES CXX)
if(NOT IS_DIRECTORY "${STRATA_ENGINE_SOURCE_ROOT}/include/strata")
  message(FATAL_ERROR "Set STRATA_ENGINE_SOURCE_ROOT to the engine checkout under measurement")
endif()
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
find_package(Threads REQUIRED)
add_executable(ingest ingest.cpp "${STRATA_ENGINE_SOURCE_ROOT}/src/engine.cpp")
target_include_directories(ingest PRIVATE "${STRATA_ENGINE_SOURCE_ROOT}/include")
target_link_libraries(ingest PRIVATE Threads::Threads)
target_compile_options(ingest PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
