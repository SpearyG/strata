cmake_minimum_required(VERSION 3.30)
project(strata VERSION 0.1.0 DESCRIPTION "Bounded partitioned telemetry engine" LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(cmake/StrataOptions.cmake)
option(STRATA_BUILD_BENCHMARKS "Build benchmark harness" ON)
option(STRATA_BUILD_EXAMPLES "Build library example" ON)

find_package(Threads REQUIRED)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <expected>\n#include <version>\n#if !defined(__cpp_lib_expected) || __cpp_lib_expected < 202202L\n#error std::expected is required\n#endif\nint main() { std::expected<int,int> v=1; return *v-1; }" STRATA_HAS_EXPECTED)
if(NOT STRATA_HAS_EXPECTED)
  message(FATAL_ERROR "A C++26 compiler with std::expected is required; see README toolchains")
endif()
if(STRATA_ENABLE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
  if(NOT ipo_supported)
    message(FATAL_ERROR "LTO requested but unavailable: ${ipo_error}")
  endif()
endif()

add_library(strata STATIC src/engine.cpp src/journal.cpp src/workload.cpp)
add_library(strata::strata ALIAS strata)
target_include_directories(strata PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(strata PUBLIC Threads::Threads)
target_compile_features(strata PUBLIC cxx_std_26)
if(STRATA_FORCE_PACK_FALLBACK)
  target_compile_definitions(strata PUBLIC STRATA_FORCE_PACK_FALLBACK=1)
endif()
strata_options(strata)

add_executable(strata_cli tools/strata.cpp)
set_target_properties(strata_cli PROPERTIES OUTPUT_NAME strata)
target_link_libraries(strata_cli PRIVATE strata::strata)
strata_options(strata_cli)
if(STRATA_BUILD_BENCHMARKS)
  add_executable(strata_bench benchmarks/strata_bench.cpp)
  target_link_libraries(strata_bench PRIVATE strata::strata)
  strata_options(strata_bench)
endif()
if(STRATA_BUILD_EXAMPLES)
  add_executable(strata_example examples/basic.cpp)
  target_link_libraries(strata_example PRIVATE strata::strata)
  strata_options(strata_example)
endif()
if(BUILD_TESTING)
  foreach(suite IN ITEMS unit runtime storage stress workload)
    add_executable(test_${suite} tests/test_${suite}.cpp)
    target_link_libraries(test_${suite} PRIVATE strata::strata)
    strata_options(test_${suite})
    add_test(NAME ${suite} COMMAND test_${suite})
    set_tests_properties(${suite} PROPERTIES TIMEOUT 90 LABELS ${suite})
  endforeach()
  add_test(NAME cli_smoke COMMAND strata_cli --events 20000 --producers 3 --workers 2 --queue-capacity 31 --batch-size 7)
  add_test(NAME cli_invalid COMMAND strata_cli --workers 0)
  add_test(NAME cli_unknown COMMAND strata_cli --not-an-option)
  set_tests_properties(cli_invalid cli_unknown PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
  set_tests_properties(cli_smoke PROPERTIES TIMEOUT 30 LABELS integration)
  add_test(NAME cli_replay COMMAND ${CMAKE_COMMAND} -DSTRATA_EXE=$<TARGET_FILE:strata_cli> -DTEST_DIRECTORY=${CMAKE_CURRENT_BINARY_DIR}/test-data -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_replay.cmake)
  set_tests_properties(cli_replay PROPERTIES TIMEOUT 65 LABELS integration)
  if(STRATA_BUILD_BENCHMARKS)
    add_test(NAME benchmark_smoke COMMAND strata_bench --events 1000 --repeats 3 --warmups 1)
    set_tests_properties(benchmark_smoke PROPERTIES TIMEOUT 30 LABELS integration)
  endif()
  if(STRATA_BUILD_EXAMPLES)
    add_test(NAME example COMMAND strata_example)
  endif()
endif()

install(TARGETS strata EXPORT strataTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT strataTargets NAMESPACE strata:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/strata)
configure_package_config_file(cmake/strataConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/strataConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/strata)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/strataConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/strataConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/strataConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/strata)
